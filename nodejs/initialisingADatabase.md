const PG_SCHEMA = `
  CREATE TABLE IF NOT EXISTS users (
    id            BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username      TEXT NOT NULL,
    name          TEXT NOT NULL,
    bio           TEXT NOT NULL DEFAULT '',
    password_hash TEXT NOT NULL,
    color         INTEGER NOT NULL DEFAULT 0,
    last_seen     BIGINT NOT NULL DEFAULT 0,
    created_at    BIGINT NOT NULL
  );
  CREATE UNIQUE INDEX IF NOT EXISTS users_username_lower ON users (lower(username));
  CREATE TABLE IF NOT EXISTS chats (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    type        TEXT NOT NULL CHECK (type IN ('private','group')),
    title       TEXT,
    private_key TEXT UNIQUE,
    created_by  BIGINT REFERENCES users(id) ON DELETE SET NULL,
    created_at  BIGINT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS chat_members (
    chat_id      BIGINT NOT NULL REFERENCES chats(id) ON DELETE CASCADE,
    user_id      BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role         TEXT NOT NULL DEFAULT 'member',
    last_read_id BIGINT NOT NULL DEFAULT 0,
    joined_at    BIGINT NOT NULL,
    PRIMARY KEY (chat_id, user_id)
  );
  CREATE TABLE IF NOT EXISTS messages (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    chat_id         BIGINT NOT NULL REFERENCES chats(id) ON DELETE CASCADE,
    sender_id       BIGINT REFERENCES users(id) ON DELETE SET NULL,
    kind            TEXT NOT NULL DEFAULT 'text',
    text            TEXT NOT NULL DEFAULT '',
    reply_to        BIGINT,
    attachment_url  TEXT,
    attachment_type TEXT,
    attachment_name TEXT,
    attachment_size BIGINT,
    edited_at       BIGINT,
    created_at      BIGINT NOT NULL
  );
  CREATE TABLE IF NOT EXISTS files (
    id         TEXT PRIMARY KEY,
    owner_id   BIGINT REFERENCES users(id) ON DELETE SET NULL,
    name       TEXT NOT NULL,
    mime       TEXT NOT NULL,
    size       BIGINT NOT NULL,
    data       BYTEA NOT NULL,
    created_at BIGINT NOT NULL
  );
  CREATE INDEX IF NOT EXISTS idx_messages_chat ON messages(chat_id, id);
  CREATE INDEX IF NOT EXISTS idx_members_user  ON chat_members(user_id);
`;

/** `?` → `$1, $2…` (вне строковых литералов) */
function toPgPlaceholders(sql) {
  let i = 0, inStr = false, out = '';
  for (const ch of sql) {
    if (ch === "'") inStr = !inStr;
    out += ch === '?' && !inStr ? `$${++i}` : ch;
  }
  return out;
}

const plain = (row) => {
  if (!row) return row;
  const o = { ...row };
  for (const k in o) if (o[k] instanceof Uint8Array && !Buffer.isBuffer(o[k])) o[k] = Buffer.from(o[k]);
  return o;
};

async function createPostgres(url) {
  const { Pool } = await import('./pg.js');
  const pool = new Pool(url, { max: Number(process.env.DB_POOL_SIZE || 5) });
  await pool.exec(PG_SCHEMA);

  const make = (runner) => ({
    async all(sql, ...params) { return (await runner(toPgPlaceholders(sql), params)).rows; },
    async get(sql, ...params) { return (await runner(toPgPlaceholders(sql), params)).rows[0]; },
    async run(sql, ...params) { return { changes: (await runner(toPgPlaceholders(sql), params)).rowCount }; },
  });

  return {
    dialect: 'postgres',
    ...make((sql, params) => pool.query(sql, params)),
    async tx(fn) {
      const conn = await pool.acquire();
      const scoped = make((sql, params) => conn.query(sql, params));
      try {
        await conn.query('BEGIN');
        const result = await fn(scoped);
        await conn.query('COMMIT');
        return result;
      } catch (err) {
        await conn.query('ROLLBACK').catch(() => {});
        throw err;
      } finally {
        pool.release(conn);
      }
    },
    close: () => pool.end(),
  };
}

export const db = await createPostgres(process.env.DATABASE_URL);
